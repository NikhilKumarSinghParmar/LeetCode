class Solution {
public:
    int minOperations(vector<vector<int>>& grid, int x) {
        vector<int> numsArray;
        int result = 0;

        // Flatten the grid into numsArray
        for (int row = 0; row < grid.size(); row++) {
            for (int col = 0; col < grid[0].size(); col++) {
                numsArray.push_back(grid[row][col]);
            }
        }

        int length = numsArray.size();

        // Partially sorts the array so that the median element is placed at the
        // middle index
        nth_element(numsArray.begin(), numsArray.begin() + length / 2,
                    numsArray.end());

        // Store the median element as the final common value to make all
        // elements equal to
        int finalCommonNumber = numsArray[length / 2];

        for (int number : numsArray) {
            // If the remainder when divided by x is different for any number,
            // return -1
            if (number % x != finalCommonNumber % x) return -1;
            // Add the number of operations required to make the current number
            // equal to finalCommonNumber
            result += abs(finalCommonNumber - number) / x;
        }

        return result;
    }
};

class Solution {
public:
    int find_gap(vector<int>& vec,int n,int k,int x){
        int gap = 0;

        for(int i=0;i<n;i++){
            gap += abs(vec[k] - vec[i]);
        }
        gap /= x;

        return gap;
    }

    bool is_valid(vector<int>& vec,int n,int x){
        for(int i=1;i<n;i++){
            if((vec[i]-vec[i-1])%x){
                return false;
            }
        }

        return true;
    }


    int minOperations(vector<vector<int>>& grid, int x) {
        int m = grid.size();
        int n = grid[0].size();
        vector<int> vec;

        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                vec.push_back(grid[i][j]);
            }
        }      

        sort(vec.begin(),vec.end());
        n = m*n;
        if(!is_valid(vec,n,x)){
            return -1;
        }
        if(n%2 || n<=2){
            int left = find_gap(vec,n,n/2,x);
            return left;
        }
        else{
            int left = find_gap(vec,n,n/2,x);
            int right = find_gap(vec,n,n/2+1,x);
            return min(left,right);
        }

        return -1;
    }
};



class Solution {
public:
    int minOperations(vector<vector<int>>& grid, int x) {
        int m = grid.size();
        int n = grid[0].size();
        int val0 = grid[0][0];
        int minval(val0),maxval(val0);
        vector<int> freq(10001,0);

        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if((grid[i][j] - val0)%x){
                    return -1;
                }

                freq[grid[i][j]]++;
                minval = min(minval,grid[i][j]);
                maxval = max(maxval,grid[i][j]);
            }
        }

        int cost(0);
        for(int i=minval+x;i<=maxval;i+=x){
            cost += (((i - minval)/x) * freq[i]);
        }

        for(int i=minval+x;i<=maxval;i+=x){
            freq[i] += freq[i-x];
        }

        int ans(cost);
        for(int i=minval+x;i<=maxval;i+=x){
            int left = freq[i-x];
            int right = freq[maxval] - freq[i-x];

            cost = cost + left - right;
            ans = min(ans,cost);
        }

        return ans;
    }
};
