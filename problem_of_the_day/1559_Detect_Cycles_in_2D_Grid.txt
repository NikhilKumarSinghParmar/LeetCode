class Solution {
private:
    int m, n;
    vector<vector<bool>> vis;
    vector<vector<int>> depth;

    vector<int> dx = {-1, 0, 1, 0};
    vector<int> dy = {0, 1, 0, -1};

    bool isValid(int x, int y) {
        return (x >= 0 && x < m && y >= 0 && y < n);
    }

    bool dfs(vector<vector<char>>& grid, int x, int y, int px, int py, 
             char ch, int d, int K) {
        
        vis[x][y] = true;
        depth[x][y] = d;

        for (int dir = 0; dir < 4; dir++) {
            int nx = x + dx[dir];
            int ny = y + dy[dir];

            if (!isValid(nx, ny)) continue;
            if (grid[nx][ny] != ch) continue;

            // Not visited → go deeper
            if (!vis[nx][ny]) {
                if (dfs(grid, nx, ny, x, y, ch, d + 1, K))
                    return true;
            }
            // Visited and not parent → cycle
            else if (nx != px || ny != py) {
                int cycle_len = d - depth[nx][ny] + 1;
                if (cycle_len >= K)
                    return true;
            }
        }

        return false;
    }

public:
    bool containsCycle(vector<vector<char>>& grid) {
        m = grid.size();
        n = grid[0].size();
        int K = 4;
        
        vis.assign(m, vector<bool>(n, false));
        depth.assign(m, vector<int>(n, 0));

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (!vis[i][j]) {
                    if (dfs(grid, i, j, -1, -1, grid[i][j], 0, K))
                        return true;
                }
            }
        }

        return false;
    }
};
